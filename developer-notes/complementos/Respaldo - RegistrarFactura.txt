import { useNavigate } from "react-router-dom";
import styles from "./RegistrarFactura.module.css";
import { useState } from "react";

export const RegistrarFactura = () => {
  // Regresar al Menú al presionar el botón de Cancelar
  const navigate = useNavigate();
  const handleCancelButton = () => {
    navigate("/facturas");
  };

  // Si es la factura es ANULADA o es una NOTA DE CREDITO, permitir describir el motivo
  const [motivo, setMotivo] = useState(Boolean);
  const handleEstatusChange = (e: React.ChangeEvent<HTMLSelectElement>) => {
    if (e.target.value !== "PROCESADA") {
      setMotivo(true);
    } else {
      setMotivo(false);
    }
  };

  // Calcular la edad del paciente
  const [edad, setEdad] = useState<number>(0);
  const handleFechaNacimiento = (e: React.ChangeEvent<HTMLInputElement>) => {
    const fechaSeleccionada = e.target.value;
    if (!fechaSeleccionada) {
      setEdad(0);
      return;
    }

    const [anioStr, mesStr, diaStr] = fechaSeleccionada.split("-");
    const anio = parseInt(anioStr, 10);
    const mes = parseInt(mesStr, 10) - 1; // Los meses van del 0 al 11 en JS
    const dia = parseInt(diaStr, 10);

    const nacimiento = new Date(anio, mes, dia);

    if (!isNaN(nacimiento.getTime())) {
      const hoy = new Date();
      let edadCalculada = hoy.getFullYear() - nacimiento.getFullYear();
      const difMes = hoy.getMonth() - nacimiento.getMonth();
      const difDia = hoy.getDate() - nacimiento.getDate();

      // Ajuste si aún no ha cumplido años este año
      if (difMes < 0 || (difMes === 0 && difDia < 0)) {
        edadCalculada--;
      }

      setEdad(edadCalculada >= 0 ? edadCalculada : 0);
    }
  };

  // Si HAY una cirugía, permitir seleccionar el tipo
  const [cirugia, setCirugia] = useState(Boolean);
  const handleCirugia = (e: React.ChangeEvent<HTMLSelectElement>) => {
    if (e.target.value === "SI") {
      setCirugia(true);
    } else {
      setCirugia(false);
    }
  };

  /* === COMPONENTE === */
  return (
    <form action="#" className={styles.form}>
      <h2 className={styles.formTitle}>Registrar Factura</h2>

      {/* === DATOS DE LA FACTURA === */}
      <fieldset>
        <legend>Datos de la Factura</legend>

        {/* Código de la Factura */}
        <div className={styles.inputWrapper}>
          <label htmlFor="idFactura">
            Código de la Factura<span className={styles.requireSymbol}>*</span>
          </label>
          <input id="idFactura" type="text" required />
        </div>

        {/* Número del Control/Nota de Crédito */}
        <div className={styles.inputWrapper}>
          <label htmlFor="idControl">
            Nro. del Control<span className={styles.requireSymbol}>*</span>
          </label>
          <input id="idControl" type="text" required />
        </div>

        {/* Fecha de Emisión */}
        <div className={styles.inputWrapper}>
          <label htmlFor="fechaEmision">
            Fecha de Emisión<span className={styles.requireSymbol}>*</span>
          </label>
          <input id="fechaEmision" type="date" required />
        </div>

        {/* Titular */}
        <div className={styles.inputWrapper}>
          <label htmlFor="titular">Titular</label>
          <input id="titular" type="text" />
        </div>

        {/* Tipo de Ingreso */}
        <div className={styles.inputWrapper}>
          <label htmlFor="tipoIngreso">Tipo de Ingreso</label>
          <select id="tipoIngreso">
            <option value="EMPRESA">Empresa</option>
            <option value="PARTICULAR">Particular</option>
          </select>
        </div>

        {/* Forma de Pago */}
        <div className={styles.inputWrapper}>
          <label htmlFor="formaPago">Forma de Pago</label>
          <select id="formaPago">
            <option value="EMPRESA">Empresa</option>
            <option value="PARTICULAR">Particular</option>
          </select>
        </div>

        {/* Suministros Hospitalarios */}
        <div className={styles.inputWrapper}>
          <label htmlFor="suminiHospitalarios">
            Suministros Hospitalarios{" "}
            <span className={styles.moneyType}>(Bs.)</span>
          </label>
          <input id="suminiHospitalarios" type="number" min={0} step={0.01} />
        </div>

        {/* Servicios Cobrables */}
        <div className={styles.inputWrapper}>
          <label htmlFor="serviCobrables">
            Servicios Cobrables <span className={styles.moneyType}>(Bs.)</span>
          </label>
          <input id="serviCobrables" type="number" min={0} step={0.01} />
        </div>

        {/* Medicamentos */}
        <div className={styles.inputWrapper}>
          <label htmlFor="medicamentos">
            Medicamentos <span className={styles.moneyType}>(Bs.)</span>
          </label>
          <input id="medicamentos" type="number" min={0} step={0.01} />
        </div>

        {/* Honorarios Médicos y Servicios Auxiliares */}
        <div className={styles.inputWrapper}>
          <label htmlFor="honoMediServiAux">
            Honorarios Médicos y Servicios Auxiliares{" "}
            <span className={styles.moneyType}>(Bs.)</span>
          </label>
          <input id="honoMediServiAux" type="number" min={0} step={0.01} />
        </div>

        {/* Tasa del Dólar */}
        <div className={styles.inputWrapper}>
          <label htmlFor="tasaDolarBCV">Tasa Dólar BCV</label>
          <input id="tasaDolarBCV" type="number" min={0} step={0.01} />
        </div>

        {/* Estatus */}
        <div className={styles.inputWrapper}>
          <label htmlFor="estatus">Estatus</label>
          <select id="estatus" onChange={handleEstatusChange}>
            <option value="">...</option>
            <option value="PROCESADA">Procesada</option>
            <option value="ANULADA">Anulada</option>
            <option value="NOTA DE CREDITO">Nota de Crédito</option>
          </select>
        </div>

        {/* Motivo (solo disponible para ANULADAS o NOTA DE CREDITO)*/}
        {motivo ? (
          <div className={styles.inputWrapper}>
            <label htmlFor="motivo">
              Motivo de la Anulación / Nota de Crédito
            </label>
            <textarea id="motivo" rows={5}></textarea>
          </div>
        ) : null}
      </fieldset>

      {/* === DATOS DEL PACIENTE === */}
      {/* TODO: Cargar los datos del paciente si el paciente ya se encuentra registrado  */}
      <fieldset>
        <legend>Datos del Paciente</legend>

        {/* Cédula del Paciente */}
        {/* El usuario la introduce y le da a Buscar, si existe, carga los demas campos de este fieldset con sus datos */}
        <div className={styles.inputWrapper}>
          <label htmlFor="cedulaPaciente">Cédula</label>
          <input id="cedulaPaciente" type="text" />
          <button className={styles.searchButton} type="button">
            Buscar
          </button>
        </div>

        {/* Nombres */}
        <div className={styles.inputWrapper}>
          <label htmlFor="nombrePaciente">Nombres</label>
          <input id="nombrePaciente" type="text" />
        </div>

        {/* Apellidos */}
        <div className={styles.inputWrapper}>
          <label htmlFor="apellidosPaciente">Apellidos</label>
          <input id="apellidosPaciente" type="text" />
        </div>

        {/* Fecha de Nacimiento */}
        <div className={styles.inputWrapper}>
          <label htmlFor="fechaNacimiento">Fecha de Nacimiento</label>
          <input
            id="fechaNacimiento"
            type="date"
            onChange={handleFechaNacimiento}
          />
        </div>

        {/* Edad */}
        <div className={styles.inputWrapper}>
          <label htmlFor="edad">Edad</label>
          <input id="edad" type="number" value={edad} readOnly />
        </div>

        {/* Dirección */}
        <div className={styles.inputWrapper}>
          <label htmlFor="direccionPaciente">Dirección</label>
          <textarea id="direccionPaciente" rows={5}></textarea>
        </div>
      </fieldset>

      {/* === DATOS DEL MÉDICO === */}
      {/* TODO: Hacer la petición para listar y asignar el médico que lo atendió */}
      <fieldset>
        <legend>Datos del Médico</legend>

        {/* Cédula del Médico */}
        <div className={styles.inputWrapper}>
          <label htmlFor="cedulaMedico">Cédula</label>
          <select id="cedulaMedico">
            <option value="">Selecciona un Médico</option>
          </select>
        </div>

        {/* Nombre del Médico */}
        {/* Esto se carga en base al médico que haya seleccionado, y como tal no se envía con el formulario ya que no hace falta, es solo para que el usuario lo tenga como referencia por si no estaba seguro de quién era la cédula */}
        <div className={styles.inputWrapper}>
          <label htmlFor="nombreMedico">Nombres</label>
          <input
            id="nombreMedico"
            className={styles.disabledInput}
            type="text"
            value="..."
            disabled
          />
        </div>
      </fieldset>

      {/* === DATOS DE LA HISTORIA === */}
      <fieldset>
        <legend>Datos de la Historia Médica</legend>

        {/* Fecha de Ingreso */}
        <div className={styles.inputWrapper}>
          <label htmlFor="fechaIngreso">Fecha de Ingreso</label>
          <input id="fechaIngreso" type="date" />
        </div>

        {/* Fecha de Egreso */}
        <div className={styles.inputWrapper}>
          <label htmlFor="fechaEgreso">Fecha de Egreso</label>
          <input id="fechaEgreso" type="date" />
        </div>

        {/* Cirugías */}
        <div className={styles.inputWrapper}>
          <label htmlFor="cirugias">Cirugía</label>
          <select id="cirugias" onChange={handleCirugia}>
            <option value="NO">No</option>
            <option value="SI">Si</option>
          </select>
        </div>

        {/* Tipo de Cirugía (solo disponible si HAY una cirugía)*/}
        {cirugia ? (
          <div className={styles.inputWrapper}>
            <label htmlFor="tipoCirugia">Tipo de Cirugía</label>
            <select id="tipoCirugia">
              <option value="">...</option>
              <option value="HOSPITALIZACION">HOSPITALIZACION</option>
              <option value="PLAN CRG">PLAN CRG</option>
              <option value="PLAN URL">PLAN URL</option>
              <option value="PLAN OFT">PLAN OFT</option>
            </select>
          </div>
        ) : null}

        {/* Diagnóstico */}
        <div className={styles.inputWrapper}>
          <label htmlFor="diagnostico">Diagnóstico</label>
          <textarea id="diagnostico" rows={5}></textarea>
        </div>
      </fieldset>

      {/* === BOTONES DE ACCIÓN === */}
      <div className={styles.buttonWrapper}>
        <button type="submit" className={styles.saveButton}>
          Registrar
        </button>
        <button
          onClick={handleCancelButton}
          type="button"
          className={styles.cancelButton}
        >
          Cancelar
        </button>
      </div>
    </form>
  );
};
